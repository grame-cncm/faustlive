#######################################
# FaustLive test suite
# This file is included by Build/CMakeLists.txt when the TESTS option is on,
# and reuses the Faust, LLVM and Qt settings computed there.
#   build : make -C Build            (or cmake --build <builddir>)
#   run   : make -C Build test       (or ctest --test-dir <builddir> --output-on-failure)
#######################################

set (TESTDIR    ${ROOT}/tests)
set (QRCFILE    ${RSRCDIR}/application.qrc)
# Mimics ~/.FaustLive-CurrentSession-<APP_VERSION> (Libs and Examples folders)
set (SESSIONDIR ${CMAKE_CURRENT_BINARY_DIR}/session)
set (CACHEDIR   ${CMAKE_CURRENT_BINARY_DIR}/SHAFolder)
set_property (DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${QRCFILE})

if (WIN32)
	set (FAUST_CORE_LIBRARY ${WFAUST_LIBRARY_DIR}/libfaust.lib)
else()
	set (FAUST_CORE_LIBRARY "-L${FAUST_LIBRARY_DIR}" -lfaust)
endif()

#######################################
# Stage the bundled libraries and examples like FaustLive does at startup
add_test (NAME stage_session
	COMMAND ${CMAKE_COMMAND} -DQRC=${QRCFILE} -DDEST=${SESSIONDIR} -P ${TESTDIR}/stage_resources.cmake)
set_tests_properties (stage_session PROPERTIES FIXTURES_SETUP session LABELS "dsp")

#######################################
# faustlive_dsp_check : libfaust + LLVM, no Qt, no audio
add_executable (faustlive_dsp_check ${TESTDIR}/faustlive_dsp_check.cpp)
set_target_properties (faustlive_dsp_check PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_include_directories (faustlive_dsp_check PRIVATE ${FAUST_INCLUDE_DIRS})
target_link_libraries (faustlive_dsp_check PRIVATE ${FAUST_CORE_LIBRARY} ${LLVM_LIBRARIES})
if (MSVC)
	target_compile_options (faustlive_dsp_check PRIVATE /permissive- "/Zc:__cplusplus")
endif()

# Every example shipped in the Qt resources must compile, run and survive the bitcode cache
file (STRINGS ${QRCFILE} EXAMPLES REGEX "<file>Examples/.*\\.dsp</file>")
foreach (ENTRY ${EXAMPLES})
	string (REGEX REPLACE ".*<file>Examples/(.*)</file>.*" "\\1" DSP ${ENTRY})
	get_filename_component (NAME ${DSP} NAME_WE)
	set (VOICES)
	file (STRINGS ${RSRCDIR}/Examples/${DSP} POLY REGEX "declare +nvoices|\"gate\"|\\[midi:")
	if (POLY)
		set (VOICES --voices 4)
	endif()
	add_test (NAME example.${NAME}
		COMMAND faustlive_dsp_check --libs ${SESSIONDIR}/Libs --examples ${SESSIONDIR}/Examples
				--cache ${CACHEDIR} ${VOICES} ${SESSIONDIR}/Examples/${DSP})
	set_tests_properties (example.${NAME} PROPERTIES FIXTURES_REQUIRED session LABELS "dsp")
endforeach()

# Extra DSP programs (tests/dsp/*.dsp), compiled from their own folder like a user file
file (GLOB EXTRA_DSPS ${TESTDIR}/dsp/*.dsp)
foreach (DSP ${EXTRA_DSPS})
	get_filename_component (NAME ${DSP} NAME_WE)
	set (VOICES)
	file (STRINGS ${DSP} POLY REGEX "declare +nvoices")
	if (POLY)
		set (VOICES --voices 4)
	endif()
	add_test (NAME dsp.${NAME}
		COMMAND faustlive_dsp_check --libs ${SESSIONDIR}/Libs --examples ${SESSIONDIR}/Examples
				--cache ${CACHEDIR} ${VOICES} ${DSP})
	set_tests_properties (dsp.${NAME} PROPERTIES FIXTURES_REQUIRED session LABELS "dsp")
endforeach()

# Every library shipped in the Qt resources must be importable from the session Libs folder
file (STRINGS ${QRCFILE} LIBS REGEX "<file.*>Libs/.*\\.lib</file>")
foreach (ENTRY ${LIBS})
	if (ENTRY MATCHES "alias=\"Libs/([^\"]+)\"")
		set (LIB ${CMAKE_MATCH_1})
	else()
		string (REGEX REPLACE ".*<file>Libs/(.*)</file>.*" "\\1" LIB ${ENTRY})
	endif()
	get_filename_component (NAME ${LIB} NAME_WE)
	set (LIBDSP ${CMAKE_CURRENT_BINARY_DIR}/libraries/import_${NAME}.dsp)
	file (WRITE ${LIBDSP} "// generated by tests/CMakeLists.txt\nimport(\"${LIB}\");\nprocess = _;\n")
	add_test (NAME library.${NAME}
		COMMAND faustlive_dsp_check --libs ${SESSIONDIR}/Libs --examples ${SESSIONDIR}/Examples ${LIBDSP})
	set_tests_properties (library.${NAME} PROPERTIES FIXTURES_REQUIRED session LABELS "dsp")
endforeach()

# Every prefix declared in stdfaust.lib (ba., fi., dx., ...) must resolve to a bundled library
add_test (NAME library.stdfaust_prefixes
	COMMAND ${CMAKE_COMMAND} -DLIBS=${SESSIONDIR}/Libs -P ${TESTDIR}/check_stdfaust.cmake)
set_tests_properties (library.stdfaust_prefixes PROPERTIES FIXTURES_REQUIRED session LABELS "dsp")

#######################################
# tst_utilities : Qt Test unit tests of the GUI-free helpers
if (QT6)
	find_package (Qt6 COMPONENTS Test QUIET)
	set (QTTEST_FOUND ${Qt6Test_FOUND})
	set (QTTESTLIB Qt6::Test)
else()
	find_package (Qt5 COMPONENTS Test QUIET)
	set (QTTEST_FOUND ${Qt5Test_FOUND})
	set (QTTESTLIB Qt5::Test)
endif()

if (QTTEST_FOUND)
	add_executable (tst_utilities
		${TESTDIR}/tst_utilities.cpp
		${SRCDIR}/Utilities/utilities.cpp
		${SRCDIR}/MenusAndDialogs/SimpleParser.cpp)
	set_target_properties (tst_utilities PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
	target_include_directories (tst_utilities PRIVATE ${SRCDIR}/Utilities ${SRCDIR}/MenusAndDialogs)
	target_link_libraries (tst_utilities PRIVATE ${QTLIBS} ${QTTESTLIB})
	add_test (NAME unit.utilities COMMAND tst_utilities)
	set_tests_properties (unit.utilities PROPERTIES LABELS "unit" ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
else()
	message (STATUS "Qt Test not found: unit tests disabled")
endif()

#######################################
# Smoke test of the FaustLive application itself (needs an audio device)
add_test (NAME app.smoke
	COMMAND ${CMAKE_COMMAND} -DAPP=$<TARGET_FILE:${faustlive}> -DWORKDIR=${CMAKE_CURRENT_BINARY_DIR}/smoke
			-DDSP=${TESTDIR}/dsp/app_smoke.dsp -DAPP_VERSION=${FAUSTLIVE_APP_VERSION}
			-P ${TESTDIR}/app_smoke.cmake)
set_tests_properties (app.smoke PROPERTIES LABELS "app" TIMEOUT 120)
